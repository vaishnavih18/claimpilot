# Added for Postgres models + Alembic migrations + LangGraph checkpoints.
# sqlalchemy/alembic: declarative models + versioned schema migrations.
# psycopg[binary]: Postgres driver - "binary" extra bundles a prebuilt libpq
# so there's no C toolchain / libpq-dev needed on either your machine or
# inside the Docker image.
# pgvector: gives SQLAlchemy a Vector column type matching the `vector`
# Postgres extension - needed for the claims table's similarity-search
# embedding column (same Titan V2 1024-dim embeddings ingest.py already
# uses for policy clauses).
# langgraph-checkpoint-postgres: LangGraph's own Postgres-backed
# checkpointer - stores agent state/conversation history as JSON blobs in
# tables it manages itself (see migrations/versions/0002_*.py for why that
# one migration looks different from the others).
# boto3: already a transitive need (ingest.py/generate.py use it for
# Bedrock) - added here explicitly because the API now also needs it, to
# fetch the RDS password from Secrets Manager at runtime instead of
# reading it from an env var in plain text.

alembic>=1.13,<1.14
psycopg[binary]>=3.1,<3.3
pgvector>=0.3,<0.4
langgraph-checkpoint-postgres>=2.0,<3.0
